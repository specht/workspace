# Ein kleines Programm, um Farbschemata zu vergleichen
import random

farben = ["Rot", "Grün", "Blau", "Lila"]
punkte = {"Ada": 42, "Linus": 17, "Grace": 23}


def begruesse(name, alter=15):
    """Begrüßt jemanden freundlich."""
    if alter >= 16 and name != "":
        print(f"Hallo {name}, schön, dass du da bist!")
    else:
        print("Hi " + name + "!")
    return len(name) * 2


for i, farbe in enumerate(farben):
    zahl = random.randint(1, 100) + i * 3.5
    begruesse(farbe.upper(), alter=zahl)
